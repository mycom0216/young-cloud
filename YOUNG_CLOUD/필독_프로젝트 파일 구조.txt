│   ├── main_window.py       # [핵심] 로그인 성공 > 사이드바 + 서브메뉴 + QStackedWidget 제어 로직

│   │   ├── login_window.py      # 로그인 및 회원가입 화면
│   │   ├── main_window.py       # 일반 사용자 메인 대시보드 (HOME, 캘린더)
│   │   ├── admin_window.py      # 관리자 모드 대시보드
│   │   ├── cloud_window.py      # 클라우드 파일함 (올리기/받기/삭제)
│   │   ├── message_window.py    # 메시지 송수신 및 상세창
│   │   └── settings_window.py   # 설정(개인정보, 메시지, 파일/블랙리스트)
│   └── utils/                   # 유틸리티 (이메일 인증, 암호화 등)
│       └── email_auth.py        # yagmail 기반 이메일 인증 모듈
│
├── .gitignore                   # Git 제외 대상 설정 (.db 파일, 캐시 등)
├── requirements.txt             # 라이브러리 목록 (PySide6, yagmail 등)
└── README.md                    # 프로젝트 소개 및 실행 방법
